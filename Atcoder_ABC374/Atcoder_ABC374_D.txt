#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
typedef vector<ll> vd;
typedef vector<int> vi;
typedef stack<ll> sd;
typedef stack<int> si;
typedef queue<ll> qd;
typedef queue<int> qi;
typedef map<int, int> mii;

double dist(double x, double y, double z, double w) {
  return sqrt((x - z) * (x - z) + (y - w) * (y - w));
}

double dmin(double x, double y) { return ((x < y) ? x : y); }

int main() {
  ios::sync_with_stdio(false);
  cin.tie(0);
  int n;
  double mintime = 100000000.0, time, s, t, p[10][2][2];
  cin >> n >> s >> t;
  for (int i = 0; i < n; i++)
    for (int j = 0; j < 2; j++)
      for (int k = 0; k < 2; k++)
        cin >> p[i][j][k];
  int a[10];
  for (int i = 0; i < n; i++)
    a[i] = i;
  do {
    for (int j = 0; j < 1 << n; j++) {
      time = dist(0, 0, p[a[0]][1 & j][0], p[a[0]][1 & j][1]) / s +
             dist(p[a[0]][1 & j][0], p[a[0]][1 & j][1], p[a[0]][1 - (1 & j)][0],
                  p[a[0]][1 - (1 & j)][1]) /
                 t;
      for (int i = 1; i < n; i++) {
        time +=
            dist(p[a[i - 1]][1 - ((1 << (i - 1) & j) >> (i - 1))][0],
                 p[a[i - 1]][1 - ((1 << (i - 1) & j) >> (i - 1))][1],
                 p[a[i]][(1 << i & j) >> i][0], p[a[i]][(1 << i & j) >> i][1]) /
                s +
            dist(p[a[i]][(1 << i & j) >> i][0], p[a[i]][(1 << i & j) >> i][1],
                 p[a[i]][1 - ((1 << i & j) >> i)][0],
                 p[a[i]][1 - ((1 << i & j) >> i)][1]) /
                t;
      }
      mintime = dmin(mintime, time);
    }
  } while (next_permutation(a, a + n));
  printf("%.8lf\n", mintime);
  return 0;
}
